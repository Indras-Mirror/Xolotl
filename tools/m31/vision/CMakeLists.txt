cmake_minimum_required(VERSION 3.24)
project(m31_vision_helper LANGUAGES C CXX)

# This is deliberately a separate CMake invocation: PR #31192's ggml ABI must
# never be linked into the Xolotl process or its pinned ggml build.
set(M31_LLAMA_SOURCE_DIR "" CACHE PATH "llama.cpp PR #31192 checkout at 6b95a990")
if(NOT M31_LLAMA_SOURCE_DIR OR NOT EXISTS "${M31_LLAMA_SOURCE_DIR}/tools/mtmd/mtmd.h")
  message(FATAL_ERROR "Set M31_LLAMA_SOURCE_DIR to the PR #31192 source checkout")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${M31_LLAMA_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE _m31_pr_sha OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT _m31_pr_sha STREQUAL "6b95a9908d2c7b242817a447708aec38e937e2c0")
  message(FATAL_ERROR "Unsupported mtmd revision: ${_m31_pr_sha}; expected PR #31192 at 6b95a990")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${M31_LLAMA_SOURCE_DIR}" diff --quiet HEAD --
  RESULT_VARIABLE _m31_pr_dirty COMMAND_ERROR_IS_FATAL ANY)
if(NOT _m31_pr_dirty EQUAL 0)
  message(FATAL_ERROR "Use an unmodified PR #31192 checkout for the vision helper")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "Build common library for the helper" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "Build mtmd library for the helper" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "Skip unrelated llama tools" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "Skip unrelated llama tests" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "Skip unrelated llama examples" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "Skip llama server" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "Skip llama unified binary" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "No network access is needed by the helper" FORCE)
set(LLAMA_SUBPROCESS ON CACHE BOOL "Enable mtmd ffmpeg video decoding" FORCE)
set(MTMD_VIDEO ON CACHE BOOL "Enable video decoding through ffmpeg" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Keep the external helper self-contained" FORCE)
add_subdirectory("${M31_LLAMA_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_executable(m31_vision_adapter vision_adapter.cpp)
target_compile_features(m31_vision_adapter PRIVATE cxx_std_20)
target_include_directories(m31_vision_adapter PRIVATE "${M31_LLAMA_SOURCE_DIR}/vendor/nlohmann")
target_link_libraries(m31_vision_adapter PRIVATE llama-common mtmd)
